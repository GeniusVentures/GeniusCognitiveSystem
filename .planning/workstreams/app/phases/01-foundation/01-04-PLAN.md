      TEST_F(GcsFFI, InitWithoutNodeReturnsNullptrGracefully)
        m_handle = gcs_init(nullptr);
        EXPECT_EQ(m_handle, nullptr);          // no crash, no exception — graceful null
        // Do NOT call gcs_shutdown on a null handle here — ShutdownNullptrIsSafeNoOp covers that.

      TEST_F(GcsFFI, InitWithExplicitPathWithoutNodeReturnsNullptrGracefully)
        Build a temp path under std::filesystem::temp_directory_path() with a unique salt;
        m_handle = gcs_init(path.c_str());
        EXPECT_EQ(m_handle, nullptr);          // explicit path does not change the option-C contract
        // No filesystem cleanup needed — init failed before any directory was created.

      TEST_F(GcsFFI, RepeatedInitCallsWithoutNodeAllReturnNullptr)
        GcsSession* h1 = gcs_init(nullptr);
        GcsSession* h2 = gcs_init(nullptr);
        GcsSession* h3 = gcs_init(nullptr);
        EXPECT_EQ(h1, nullptr);
        EXPECT_EQ(h2, nullptr);
        EXPECT_EQ(h3, nullptr);
        // m_handle stays nullptr — TearDown's conditional shutdown is a no-op.

      TEST_F(GcsFFI, StringFreeNullptrDoesNotCrash)
        gcs_string_free(nullptr);
        SUCCEED();

        gcs_shutdown(nullptr);
        SUCCEED();

      TEST_F(GcsFFI, JoinTopicValidatesArgumentsWithoutNode)
