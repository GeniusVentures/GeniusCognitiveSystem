```proto
syntax = "proto3";

package gcs.chat;

// Message role taxonomy (D-18).
enum MessageRole {
  MESSAGE_ROLE_UNSPECIFIED = 0;
  MESSAGE_ROLE_USER_SELF = 1;
  MESSAGE_ROLE_USER_PEER = 2;
  MESSAGE_ROLE_ASSISTANT = 3;
  MESSAGE_ROLE_SYSTEM = 4;
}

// Message state taxonomy (D-19).
enum MessageState {
  MESSAGE_STATE_UNSPECIFIED = 0;
  MESSAGE_STATE_PENDING = 1;
  MESSAGE_STATE_STREAMING = 2;
  MESSAGE_STATE_THINKING = 3;
  MESSAGE_STATE_COMPLETE = 4;
  MESSAGE_STATE_ERROR = 5;
}

// Authoritative message record. This proto type IS the C++ half (D-24 reinterpreted by D-26):
// no chat_message_state.hpp.jinja2 + nlohmann_json — the generated class is reused directly.
message ChatMessageState {
  string id = 1;
  string room_topic = 2;
  MessageRole role = 3;
  MessageState state = 4;
  string text = 5;
  int64 timestamp = 6;
}

// Pushed room-list event (D-21/D-26): the set of topics the session is in.
message RoomList {
  repeated string room_topic = 1;
}

// Pushed readiness event (D-26): true once init + smoke-topic pre-join completes.
message Readiness {
  bool ready = 1;
}

// Envelope for every C++ → Dart pushed event (D-26 push-not-pull).
message GcsEvent {
  oneof payload {
    ChatMessageState message = 1;
    RoomList room_list = 2;
    Readiness readiness = 3;
  }
}
```
