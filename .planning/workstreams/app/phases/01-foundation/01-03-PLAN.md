and the `gcs_ffi` SHARED library exposing the minimal topic pub/sub C ABI (D-27/D-29):
init / shutdown / publish / subscribe, payloads as codec-tagged bytes.

Purpose: the C++↔Dart data plane is symmetric in-process topic pub/sub (D-27) — Dart publishes
chat-shaped `GcsCommand` envelopes to the command topic; C++ owns all state (D-04), stamps the
authoritative `ChatMessageState`, and pushes `GcsEvent` envelopes (message/room-list/readiness/
raw-error-string) to the subscribed Dart NativePort. No pull calls, no per-feature call functions,
no `char*` payloads.

implements command dispatch + event push (with a `PostToDart` seam for Plan 05's Dart_PostCobject
wiring). The Dart-side decode/publish spike is Plan 05; the C ABI is stable so ffigen can run now.

Output: `gcs_proto` static target + `gcs_ffi` SHARED target exporting the session API. Smoke-tested
by Plan 04's test_gcs_ffi.cpp; Dart-side spike in Plan 05.

<!-- The exact proto contract (write this file verbatim as src/proto/gcs_chat.proto).
     D-29: authored .proto IS the canonical schema source; messages broken up; append-only from
     day one (D-24/ai-boss D-06/D-07) — only add fields/oneof variants, never renumber. -->
```proto
syntax = "proto3";

package gcs.chat;

// Wire codec bound to a store at creation (D-29: per-store codec choice).
enum Codec {
  CODEC_UNSPECIFIED = 0;
  CODEC_PROTOBUF = 1;
  CODEC_JSON = 2;
}

// Store/session configuration carried by gcs_init bytes (D-29: codec bound at creation).
message GcsConfig {
  string db_path = 1;
  Codec codec = 2;
}

// Message role taxonomy (D-18).
enum MessageRole {
  MESSAGE_ROLE_UNSPECIFIED = 0;
  MESSAGE_ROLE_USER_SELF = 1;
  MESSAGE_ROLE_USER_PEER = 2;
  MESSAGE_ROLE_ASSISTANT = 3;
  MESSAGE_ROLE_SYSTEM = 4;
}

// Message state taxonomy (D-19).
enum MessageState {
  MESSAGE_STATE_UNSPECIFIED = 0;
  MESSAGE_STATE_PENDING = 1;
  MESSAGE_STATE_STREAMING = 2;
  MESSAGE_STATE_THINKING = 3;
  MESSAGE_STATE_COMPLETE = 4;
  MESSAGE_STATE_ERROR = 5;
}

// Thin chat-shaped struct Dart publishes for a send (D-27: data-only; C++ stamps authority).
message SendTextCommand {
  string room_topic = 1;
  string text = 2;
}

// Join a chat room topic (C++ joins GossipSub and pushes the updated RoomList).
message JoinTopicCommand {
  string room_topic = 1;
}

// Envelope for every Dart -> C++ command publish (D-27: commands are topic publishes).
message GcsCommand {
  oneof payload {
    JoinTopicCommand join_topic = 1;
    SendTextCommand send_text = 2;
  }
}

// Authoritative message record. This proto type IS the C++ half (D-24 reinterpreted by D-26):
// C++ stamps id/timestamp/role/state; Dart renders.
message ChatMessageState {
  string id = 1;
  string room_topic = 2;
  MessageRole role = 3;
  MessageState state = 4;
  string text = 5;
  int64 timestamp = 6;
}

// Pushed room-list event (D-21/D-26): the set of topics the session is in.
message RoomList {
  repeated string room_topic = 1;
}

// Pushed readiness event (D-26): true once init + smoke-topic pre-join completes.
message Readiness {
  bool ready = 1;
}

// Pushed raw error string (D-29: errors cross FFI as raw strings on the push port).
message ErrorNotice {
  string message = 1;
}

// Envelope for every C++ -> Dart pushed event (D-26 push-not-pull).
message GcsEvent {
  oneof payload {
    ChatMessageState message = 1;
    RoomList room_list = 2;
    Readiness readiness = 3;
    ErrorNotice error = 4;
  }
}
```
